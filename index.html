#!/usr/bin/env python3 # auto_release.py - 自动创建GitHub Release import os import subprocess import requests from datetime import datetime # 配置 REPO = "your-username/your-repo" # 修改为你的仓库 GITHUB_TOKEN = os.environ.get("GITHUB_TOKEN") def get_latest_tag(): """获取最新的git tag""" try: tags = subprocess.check_output( ["git", "tag", "--sort=-v:refname"], text=True ).strip().split('\n') return tags[0] if tags else None except subprocess.CalledProcessError: return None def create_release(tag_name, release_name, body): """使用GitHub API创建Release""" url = f"https://api.github.com/repos/{REPO}/releases" headers = { "Authorization": f"token {GITHUB_TOKEN}", "Accept": "application/vnd.github.v3+json" } data = { "tag_name": tag_name, "target_commitish": "main", "name": release_name, "body": body, "draft": False, "prerelease": False } response = requests.post(url, headers=headers, json=data) if response.status_code == 201: print(f"✅ Release '{release_name}' created successfully!") return response.json() else: print(f"❌ Failed to create release: {response.text}") return None if __name__ == "__main__": version = input("输入版本号 (例如: v1.0.0): ") release_name = f"Release {version}" body = f""" ## 版本 {version} ### 新增功能 - ### 修复问题 - ### 已知问题 - 发布日期: {datetime.now().strftime('%Y-%m-%d')} """ print(f"\n创建Release: {release_name}") create_release(version, release_name, body)
